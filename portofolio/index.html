# Bubble Sort
my_array = [7, 3, 9, 12, 11]
n = len(my_array)
for i in range(n-1):
    swapped = False
    for j in range(n-i-1):
        if my_array[j] > my_array[j+1]:
            my_array[j], my_array[j+1] = my_array[j+1], my_array[j]
            swapped = True
    if not swapped:
        break 
print("Bubble Sort array:", my_array)

# Selection Sort (Optimized)
my_array2 = [64, 34, 25, 5, 22, 11, 90, 12]
n = len(my_array2)
for i in range(n-1):
    min_index = i 
    for j in range(i+1, n):
        if my_array2[j] < my_array2[min_index]:
            min_index = j
    my_array2[i], my_array2[min_index] = my_array2[min_index], my_array2[i]
print("Selection Sort array:", my_array2)

# Merge Sort Array (Ascending)
def mergeSort(arr):
    if len(arr) <= 1:
        return arr
    
    mid = len(arr) // 2
    L = arr[:mid]
    R = arr[mid:]

    sortedleft = mergeSort(L)
    sortedright = mergeSort(R)
   
    return merge(sortedleft, sortedright)

def merge(left, right):
    result = []
    i, j = 0, 0
    
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1

    result.extend(left[i:])
    result.extend(right[j:])
    
    return result

unsortedArr = [3, 7, 6, -10, 23.5, 55, -13]
sortedArr = mergeSort(unsortedArr)
print("Sorted Array:", sortedArr)

# Node Class untuk Linked List
class Node:
    def __init__(self, data):
        self.data = data
        self.next = None

# Queue dengan Linked List (FIFO)
class Queue:
    def __init__(self):
        self.front = None
        self.rear = None
        self.size = 0

    def enqueue(self, item):
        new_node = Node(item)
        if self.isEmpty():
            self.front = self.rear = new_node
        else:
            self.rear.next = new_node
            self.rear = new_node
        self.size += 1

    def dequeue(self):
        if not self.isEmpty():
            dequeued_item = self.front.data
            self.front = self.front.next
            self.size -= 1
            if self.front is None:
                self.rear = None
            return dequeued_item
        return None

    def isEmpty(self):
        return self.front is None

q = Queue()
q.enqueue(1)
q.enqueue(2)
print("Dequeue:", q.dequeue()) # Output: 1
